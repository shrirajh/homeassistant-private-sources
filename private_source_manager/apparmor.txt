#include <tunables/global>

profile private_source_manager flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  network,
  capability,
  signal (send) set=(kill,term,int,hup,cont),
  ptrace (trace,read) peer=@{profile_name},

  # AppArmor matches on the resolved path, so every symlinked entry point needs a
  # rule covering its target, not the link. tools/check_apparmor.py enforces this.
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /usr/libexec/** ix,
  /command/** ix,
  /package/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /etc/s6-overlay/** rix,

  # /usr/bin/bashio is a symlink to /usr/lib/bashio/bashio, and bashio sources the
  # rest of the directory, so it needs read as well as execute.
  /usr/lib/bashio/** rix,

  # The askpass helper git runs to collect a token.
  /opt/psm/psm/bin/** rix,

  # Read broadly, write only where the add-on legitimately needs to.
  # m is required as well as r: loading a shared library is an executable mmap, and
  # without it the dynamic linker cannot map libpython, libcrypto, or the compiled
  # extension modules under site-packages.
  /** mr,
  /data/** rw,
  /homeassistant/** rw,
  /addons/** rw,
  /tmp/** rw,
  /run/** rwk,
  /var/** rw,
  /dev/** rw,
  /opt/psm/** r,

  # Nothing here should ever rewrite the container's own configuration.
  deny /etc/passwd w,
  deny /etc/shadow rw,
  deny /etc/ssh/** w,
  deny /root/** w,
}
